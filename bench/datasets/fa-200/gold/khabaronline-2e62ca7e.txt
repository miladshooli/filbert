این سیاست‌ها بر محتوای تولیدی و بازنشرشده در وب‌سایت و حساب‌های رسمی خبرآنلاین حاکم است و خبرنگاران، نویسندگان، دبیران، سردبیران، عکاسان، تولیدکنندگان چندرسانه‌ای و همکاران شبکه‌های اجتماعی موظف به رعایت ضوابط مرتبط با وظایف خود هستند. ضوابط شفافیت محتوای تجاری نیز برای واحدهای مرتبط با تبلیغات لازم‌الاجراست.

بخش اول: درباره خبرآنلاین و مشی رسانه‌ای آن

۱. مأموریت ما

خبرآنلاین فعالیت خود را از سال ۱۳۸۷ آغاز کرده و به‌عنوان یک رسانه خبری خصوصی، مأموریت خود را تولید، انتخاب، بررسی، تحلیل و انتشار اخبار و گزارش‌های معتبر درباره مهم‌ترین رویدادهای ایران و جهان می‌داند.

خبرآنلاین رسانه‌ای تحلیلی‌ـ‌خبری است. در کنار اطلاع‌رسانی بهنگام، می‌کوشد رویدادها را در زمینه سیاسی، اجتماعی، اقتصادی، فرهنگی و انسانی آن‌ها توضیح دهد و از بازنشر سطحی خبرها فراتر رود.

هدف، کمک به افزایش آگاهی عمومی، تقویت حق پرسشگری شهروندان و فراهم‌کردن زمینه نقد کارشناسانه و مسئولانه عملکرد نهادها، سیاست‌ها و تصمیم‌های عمومی است.

درآمد خبرآنلاین عمدتاً از تبلیغات نمایشی، رپورتاژ آگهی و محتوای دارای حمایت مالی تأمین می‌شود. روابط تجاری تابع ضوابط بند «تبلیغات و محتوای دارای حمایت مالی» در همین سند هستند و بر تصمیم‌های تحریریه اثر نمی‌گذارد.

۳. استقلال حرفه‌ای و مسئولیت عمومی

خبرآنلاین در چارچوب قوانین کشور، خود را متعهد به استقلال حرفه‌ای، دقت، انصاف، شفافیت و پاسخ‌گویی می‌داند.

استقلال حرفه‌ای به معنای نداشتن مشی رسانه‌ای نیست؛ به معنای آن است که منافع سیاسی، اقتصادی، تبلیغاتی یا شخصی موجب تحریف واقعیت، پنهان‌کردن اطلاعات ضروری یا انتشار ادعاهای بی‌پشتوانه نمی‌شوند.

تصمیم درباره انتخاب، انتشار، تیترگذاری، اصلاح یا حذف محتوای خبری بر عهده مسئولان تحریریه است. روابط تجاری و تبلیغاتی و فشارهای بیرونی مبنای این تصمیم‌ها قرار نمی‌گیرند. اقدام بر اساس الزام قانونی، حفظ حریم خصوصی یا رفع آسیب، تابع ضوابط بخش پنجم این سند است.

۴. نقد، تنوع دیدگاه و تفکیک خبر از نظر

نقد کارشناسانه دستگاه‌ها، مسئولان، احزاب، جریان‌ها و تصمیم‌های عمومی بخشی از وظیفه رسانه‌ای خبرآنلاین است. این نقد بر داده، استدلال، انصاف و زبان حرفه‌ای استوار و از توهین، تحقیر، افترا و تحریف فاصله می‌گیرد.

در موضوعات محل اختلاف، دیدگاه‌های مرتبط و معتبر متناسب با اهمیت و پشتوانه آن‌ها بازتاب می‌یابند. انصاف به معنای هم‌وزن‌کردن ادعای فاقد شواهد با اطلاعات مستند نیست.

خبرآنلاین میان خبر، گزارش، تحلیل، یادداشت، گفت‌وگو، راستی‌آزمایی و محتوای تبلیغاتی تمایز قائل است. نوع محتوا از طریق برچسب، معرفی نویسنده یا توضیح روشن برای مخاطب قابل تشخیص است.

انتشار یادداشت یا گفت‌وگو لزوماً به معنای تأیید همه دیدگاه‌های مطرح‌شده در آن نیست؛ با این حال، تحریریه در قبال دقت در بازنمایی محتوا، امانت در نقل‌قول و رعایت این سیاست‌ها مسئول است.

۵. دقت پیش از سرعت

سرعت در اطلاع‌رسانی اهمیت دارد، اما جایگزین دقت نیست. در خبرهای فوری، اصل بر انتشار اطلاعات بررسی‌شده، انتساب روشن ادعاها، بیان محدودیت اطلاعات و به‌روزرسانی مسئولانه است.

اعتماد مخاطب با رفتار روزانه تحریریه ساخته می‌شود: خبر دقیق، تیتر وفادار به محتوا، منبع روشن، اصلاح آشکار خطا و احترام به حقوق انسان‌ها.

۱. دقت و حقیقت‌جویی

اطلاعات خبری تا حد متناسب با اهمیت و حساسیت موضوع بررسی می‌شوند. خبرنگار و دبیر میان واقعیت احرازشده، ادعای منبع، تحلیل و پیش‌بینی تفاوت می‌گذارند و این تفاوت را در محتوای منتشرشده حفظ می‌کنند.

اطلاعات ضروری به‌گونه‌ای حذف یا گزینش می‌شوند که برداشت مخاطب از واقعیت تغییر نکند.

۲. امانت در نقل‌قول و استفاده از منابع

نقل‌قول، سند، داده، تصویر و آثار دیگران با حفظ معنا، زمینه و انتساب درست استفاده می‌شوند.

کوتاه‌کردن یا ویرایش زبانی نقل‌قول موضع گوینده، موضوع سخن یا درجه قطعیت آن را تغییر نمی‌دهد. ترجمه نیز به متن اصلی وفادار است.

ذکر منبع را در محتواهایی که انتشار می‌دهد رعایت می‌کند با این حال بازنشر یک ادعا از رسانه دیگر نیز مسئولیت بررسی و بازنمایی دقیق آن را از خبرآنلاین سلب نمی‌کند.

۳. انصاف و حق پاسخ

در گزارش‌های حاوی اتهام، ادعای قصور یا نقد مشخص درباره اشخاص و نهادها، باید پیش از انتشار برای دریافت پاسخ اقدام شود و فرصت متناسب با فوریت و پیچیدگی موضوع در اختیار طرف مربوط قرار می‌گیرد.

پاسخ منصفانه و بدون تحریف بازتاب می‌یابد. دریافت پاسخ به معنای الزام به انتشار توهین، اطلاعات شخصی نامرتبط یا ادعاهای تازه و بررسی‌نشده نیست.

جزئیات اجرایی این اصل در بند «حق پاسخ» در بخش چهارم آمده است.

۴. اصلاح شفاف خطا

خطاهای مؤثر بر معنای خبر، اعتبار اشخاص یا تصمیم مخاطبان پس از احراز، بدون تأخیر غیرضروری اصلاح می‌شوند. اصلاح چنین خطاهایی صرفاً با تغییر پنهان متن انجام نمی‌گیرد.

خبرآنلاین میان «اصلاح خطا»، «افزودن اطلاعات تازه» و «ویرایش نگارشی» تفاوت می‌گذارد و متناسب با نوع تغییر، توضیح لازم را در اختیار مخاطب قرار می‌دهد.

۵. حریم خصوصی و کرامت انسانی

انتشار اطلاعات شخصی ضرورت خبری و منفعت عمومی قابل دفاع دارد. کنجکاوی عمومی، به‌تنهایی، توجیه‌کننده نقض حریم خصوصی نیست.

در پوشش محتوای مربوط به اخبار کودکان، قربانیان خشونت، بازماندگان حوادث و گروه‌های آسیب‌پذیر، ملاحظات حمایتی رعایت می‌شود. انتشار نام، تصویر یا جزئیاتی که به شناسایی و آسیب‌دیدن این افراد منجر می‌شود، مورد بررسی ویژه سردبیری قرار می‌گیرد.

۶. پرهیز از تبعیض و آسیب غیرضروری

زبان خبر تحقیرآمیز، نفرت‌انگیز یا مبتنی بر کلیشه‌های قومی، مذهبی، جنسیتی، سنی یا مرتبط با معلولیت نیست.

ویژگی‌های شخصی افراد تنها زمانی ذکر می‌شوند که با موضوع خبر ارتباط معنادار داشته باشند. در پوشش اتهام‌ها و پرونده‌های قضایی، اصل برائت و تفاوت میان متهم، محکوم و محکومیت قطعی رعایت می‌شود.

۷. مدیریت تعارض منافع

اعضای تحریریه منافع مالی، خانوادگی، سازمانی یا شخصی مؤثر بر پوشش خبری خود را به مسئول مستقیم اعلام می‌کنند. بر اساس ماهیت تعارض، دبیر یا سردبیر درباره واگذاری موضوع به فرد دیگر، نظارت مستقل یا افشای رابطه مرتبط برای مخاطب تصمیم می‌گیرد.

دریافت وجه، هدیه یا امتیاز در برابر انتشار، عدم انتشار یا تغییر جهت محتوای تحریریه ممنوع است. سفرهای حمایتی و همکاری‌های بیرونی مرتبط با حوزه خبری پیشاپیش اعلام و بررسی می‌شوند.

۸. استفاده مسئولانه از فناوری

فناوری و هوش مصنوعی ابزار کمکی‌اند، نه منبع مستقل حقیقت و نه جایگزین مسئولیت انسانی. مسئولیت صحت، انصاف و رعایت حقوق اشخاص در محتوای منتشرشده با تحریریه باقی می‌ماند.

۹. پاسخ‌گویی به مخاطب

گزارش خطا، درخواست اصلاح، نقد و جوابیه از مسیرهای مشخص دریافت و بررسی می‌شوند. مخاطب برای اعلام خطا موظف به اثبات سوءنیت رسانه نیست؛ ارائه نشانی مطلب و توضیح مسئله برای آغاز بررسی کفایت می‌کند.

۱. اصل تطابق

تیتر، روتیتر، زیرتیتر، لید، متن، تصویر شاخص و توضیح تصویر، بازنمایی سازگار و غیرگمراه‌کننده‌ای از محتوا ارائه می‌کند.

ادعای اصلی تیتر در لید یا بخش ابتدایی مطلب روشن می‌شود و در متن، پشتوانه کافی دارد. تیتر از نظر دامنه، شدت یا قطعیت از شواهد خبر فراتر نمی‌رود.

این اصل درباره عنوان صفحه در نتایج جست‌وجو، توضیحات متا، عنوان و توضیح اشتراک‌گذاری در شبکه‌های اجتماعی، اعلان‌ها و دیگر معرفی‌های محتوا نیز برقرار است. یکسان‌بودن لفظی همه این اجزا الزامی نیست، اما تفاوت آن‌ها به تعارض معنایی منجر نمی‌شود.

در بسیاری از فیدهای خبری، نتایج جست‌وجو، اعلان‌ها و سطوح پیشنهادی (ازجمله Google Discover)، مخاطب ممکن است پیش از کلیک تنها تیتر و تصویر شاخص یا خلاصه‌ای کوتاه را ببیند. ازاین‌رو تیتر و تصویر به‌تنهایی برداشتی درست از اصل خبر می‌دهد، حتی اگر مخاطب لید و متن را نخواند. یکسان‌بودن لفظی این اجزا الزامی نیست، اما در ادعا، مصداق، زمان، شرط و درجه قطعیت هم‌معنا می‌مانند.

۲. حفظ انتساب و درجه قطعیت

- تفاوت میان «ادعا شد»، «پیشنهاد شد»، «در حال بررسی است»، «پیش‌بینی شد»، «اعلام شد»، «تصویب شد» و «اجرا شد» حفظ می‌شود.

- انتشار سخن یک مقام یا کارشناس، به معنای تأیید مستقل محتوای آن نیست. هرگاه حذف نام یا جایگاه گوینده، ادعا را به‌صورت واقعیتی تأییدشده جلوه دهد، انتساب در تیتر نیز مشخص می‌شود.

همچنین،

پیشنهاد به‌عنوان تصمیم نهایی معرفی نمی‌شود؛

تصویب را معادل آغاز اجرا نمی‌داند؛

پیش‌بینی به‌جای نتیجه قطعی نمی‌نشیند؛

تصمیم مربوط به یک گروه به همه افراد تعمیم داده نمی‌شود؛

موضوع یا مخاطب اصلی یک اظهارنظر را تغییر نمی‌دهد.

۳. پرهیز از پنهان‌کردن اطلاعات کلیدی

- تیتر موضوع و مهم‌ترین یافته خبر را صریح بیان می‌کند. حذف عمدی اطلاعات تعیین‌کننده برای ایجاد کنجکاوی مصنوعی ممنوع است.

- برخی خبرها مستقیماً بر تصمیم‌های مالی و معیشتی مخاطب اثر می‌گذارند، از جمله خبرهای قیمت روز، حقوق و دستمزد، بازنشستگی، یارانه، کالابرگ، وام، مسکن و خودرو. در این خبرها قاعده سخت‌گیرانه‌تری اجرا می‌شود:

اگر عدد، نرخ یا مبلغ هسته اصلی خبر است، مهم‌ترین عدد مرتبط در تیتر می‌آید.

اگر نبود گروه مشمول یا وضعیت تصمیم (پیشنهاد، تصویب یا اجرا) برداشت مخاطب را تغییر دهد، در تیتر روشن می‌شود.

هرگاه تاریخ، زمان مرجع یا شرط اصلی برای فهم درست خبر تعیین‌کننده باشد، در تیتر ذکر می‌شود؛ در مواردی که محدودیت معقول طول تیتر مانع ذکر جزئیات تکمیلی باشد، اصل شرط در تیتر و جزئیات در نخستین جمله لید می‌آید.

استفاده از ضمیر یا اشاره مبهم، مانند «این افراد»، «این خودروها»، «این استان‌ها» یا «این رقم»، برای پنهان‌کردن مصداق یا عدد ممنوع است، مگر اینکه مرجع آن در خود تیتر آمده باشد.

پرسش‌هایی مانند «چند شد؟»، «چقدر است؟»، «کیلویی چند؟» یا «کدام گروه؟» به‌جای اعلام پاسخی که در خبر وجود دارد نمی‌آید.

۴. تیترهای سؤالی

تیتر سؤالی به‌خودی‌خود ممنوع نیست. استفاده از آن در مطالب توضیحی، تحلیلی، آموزشی یا راستی‌آزمایی، هنگامی استفاده می‌شود که پرسش واقعی و مشخصی را طرح کند و متن پاسخی مستند یا توضیح روشنی درباره محدودیت پاسخ ارائه دهد.

تیتر سؤالی:

پاسخ ساده و قطعیِ موجود در خبر را عمداً پنهان نمی‌کند؛

ادعای بی‌پشتوانه یا اتهام را در قالب پرسش القا نمی‌کند؛

وقوع رویدادی تأییدنشده را محتمل‌تر از شواهد موجود نشان نمی‌دهد؛

وعده پاسخی نمی‌دهد که در متن وجود ندارد.

۵. واژگان هیجانی و برچسب «فوری»

تعبیرهایی مانند «شوک بزرگ»، «باورنکردنی»، «سکته بازار» و «منفجر شد» جایگزین توصیف دقیق واقعیت نمی‌شوند.

واژه‌های ارزش‌گذار یا توصیف‌کننده شدت رویداد فقط در صورت تناسب با شواهد و زمینه خبر قابل استفاده‌اند. نقل چنین تعبیری از یک منبع نیز وسیله‌ای برای القای برداشت گمراه‌کننده نیست.

برچسب «فوری» صرفاً برای تحولی تازه، مهم و زمان‌حساس و با تأیید دبیر مسئول قابل استفاده است؛ نه برای افزایش کلیک، بازنشر خبر قدیمی یا برجسته‌سازی محتوای معمول.

۶. اعداد، تاریخ‌ها و اخبار قیمت

هرگاه عدد، مبلغ، تاریخ یا گروه مشمول، هسته اصلی خبر باشد، تیتر مهم‌ترین داده مرتبط را منعکس می‌کند. درج همه اعداد و جزئیات در تیتر ضروری نیست زیرا موجب طولانی‌شدن ناموجه یا ابهام آن می‌شود.

در خبرهای قیمت و بازار:

نوع بازار، کالا، واحد پول و واحد اندازه‌گیری روشن است.

منبع، زمان ثبت نرخ و ماهیت آن، مانند قیمت رسمی، پیشنهادی، معامله‌شده یا میانگین، در متن مشخص می‌شود.

قیمت یک فروشنده یا یک نمونه به کل بازار تعمیم داده نمی‌شود.

در گزارش نرخ روز، تیتر نرخ شاخص یا تغییر اصلی را بیان می‌کند و تاریخ مرجع گزارش برای مخاطب روشن است.

در مقایسه قیمت‌ها، مبنای مقایسه و بازه زمانی ذکر می‌شود.

تعابیر تحلیلی مانند «مقاومت قیمتی» پشتوانه و انتساب روشن دارد و جایگزین اعلام نرخ نمی‌شوند.

در گزارش‌های قیمت روزانه که در طول روز به‌روزرسانی می‌شوند، زمان آخرین به‌روزرسانی در بالای مطلب نمایش داده می‌شود و تغییرهای مهم نرخ در همان روز به‌صورت خلاصه در متن ذکر می‌شود.

۷. وعده‌های تیتر و محتوای صفحه

اگر تیتر از جدول، فهرست اسامی، سند، تصویر یا ویدئو خبر می‌دهد، آن محتوا واقعاً در صفحه وجود دارد و برای مخاطب قابل دسترس است.

عبارت‌هایی مانند «جزئیات» یا «ببینید» جایگزین اصل خبر نمی‌شوند یا انتظار دریافت اطلاعاتی را ایجاد نمی‌کنند که در صفحه ارائه نمی‌شود.

نقطه‌چین، نقل‌قول ناتمام و حذف بخش تعیین‌کننده سخن برای تغییر معنا یا ایجاد تعلیق مصنوعی به کار نمی‌روند.

۸. خبرهای قدیمی و بازنشر

محتوای قدیمی با تغییر تاریخ، تیتر یا تصویر به‌عنوان رویدادی تازه عرضه نمی‌شود.

بازنشر به مناسبت یک رویداد جدید با توضیح روشن درباره زمان و زمینه محتوای اصلی همراه است. ویرایش نگارشی یا تغییرات فنی، به‌تنهایی، به معنای به‌روزرسانی خبری نیست.

۱. ارزیابی منابع و سطح اطمینان

اعتبار خبر صرفاً با رسمی‌بودن منبع یا تعداد منابع تعیین نمی‌شود. دسترسی مستقیم، استقلال منابع از یکدیگر، اصالت سند، سابقه اعتبار، امکان بررسی و وجود شواهد متعارض هم‌زمان سنجیده می‌شوند.

چند رسانه که همگی یک ادعای واحد را بازنشر کرده‌اند، چند منبع مستقل محسوب نمی‌شوند. سند رسمی نیز از نظر اصالت، تاریخ، دامنه اعتبار و معنای دقیق بررسی می‌شود.

عبارت «در حال بررسی» مجوزی برای انتشار هر ادعای تأییدنشده نیست. در تیتر نیز از تکرار برجسته ادعایی که متن قادر به تأیید آن نیست پرهیز می‌شود.

۲. مسئولیت‌های پیش از انتشار

خبرنگار یا تهیه‌کننده محتوا مسئول بررسی اولیه، ثبت منابع، امانت در نقل‌قول و اعلام ابهام‌ها یا تعارض منافع است.

دبیر سرویس مسئول بررسی کفایت شواهد، تطابق تیتر و متن، درجه قطعیت، تصویر و رعایت ضوابط انتشار است.

بازبین دوم در مطالب پرریسک، فردی غیر از تنظیم‌کننده اولیه است که ادعای اصلی و بازنمایی آن را مستقل از بررسی نخست کنترل می‌کند.

سردبیر یا مسئول ارشد کشیک درباره انتشار موارد حساس، اختلاف‌های حرفه‌ای، استثناهای موجه و اصلاحات مهم تصمیم می‌گیرد.

مطالب پرریسک شامل خبرهایی با اثر مستقیم بر تصمیم‌های مالی، سلامت، امنیت، حیثیت اشخاص یا حقوق گروه‌های آسیب‌پذیر است. خبرهای حساس درباره قیمت، حقوق، بازنشستگی، یارانه، وام و تصمیم‌های اجرایی نیز مشمول بازبینی دوم‌اند.

کنترل انتشار بر پایه این زنجیره انجام می‌شود:

ادعای تیتر ← توضیح لید ← شواهد متن ← منبع ← درجه قطعیت

۳. خبر فوری

تحریریه برای انتشار خبر فوری:

منبع اولیه و زمان دریافت اطلاعات را ثبت می‌کند.

اهمیت عمومی و فوریت واقعی خبر را می‌سنجد.

برای گزاره‌ای که قرار است منتشر شود، شواهد متناسب فراهم می‌کند.

فقط اطلاعات بررسی‌شده را منتشر و ادعاها را به منبع منتسب می‌کند.

محدودیت‌ها و اولیه‌بودن اطلاعات را روشن می‌سازد.

اطلاعات تازه را با زمان به‌روزرسانی به صفحه می‌افزاید.

در صورت احراز خطا، اصلاحیه قابل مشاهده منتشر می‌کند.

فوریت به حذف انتساب، قطعی‌نمایی اطلاعات ناتمام یا کنارگذاشتن بررسی ضروری منجر نمی‌شود. در مطالب فوری پرریسک، بازبینی مسئول ارشد کشیک پیش از انتشار انجام می‌شود.

۴. منابع ناشناس

استفاده از منبع ناشناس استثناست و مستلزم تحقق هم‌زمان شرایط زیر است:

موضوع اهمیت عمومی قابل دفاع داشته باشد.

منبع به اطلاعات مورد ادعا دسترسی مستقیم و مرتبط داشته باشد.

برای محفوظ‌ماندن هویت او دلیل موجه وجود داشته باشد.

سردبیر یا دبیر ارشد مسئول، هویت و میزان دسترسی منبع را بداند.

ادعا با سند، منبع مستقل یا شواهد معتبر پشتیبانی شود.

انگیزه احتمالی منبع و خطر سوءاستفاده از رسانه بررسی شود.

در متن، بدون افشای هویت، نسبت منبع با موضوع و علت محرمانگی تا حد ایمن توضیح داده می‌شود. اتهام علیه اشخاص یا نهادها صرفاً بر ادعای یک منبع ناشناس استوار نیست.

اطلاعات شناسایی منابع محرمانه فقط در اختیار افراد ضروری قرار می‌گیرد و در ابزارها یا سامانه‌های نامطمئن وارد نمی‌شود.

۵. حق پاسخ

در گزارش‌های حاوی اتهام یا نقد مشخص، تلاش برای دریافت پاسخ واقعی و قابل ثبت است. پرسش‌های اصلی، شیوه تماس، زمان تماس و مهلت پاسخ نزد تحریریه ثبت می‌شوند.

اگر پاسخی تا زمان انتشار دریافت نشود، متن وضعیت واقعی پیگیری را توضیح می‌دهد. عبارت «آماده انتشار پاسخ هستیم» جایگزین تلاش پیشینی برای تماس نیست.

انتشار پیش از دریافت پاسخ، در موارد دارای فوریت و منفعت عمومی روشن، با تأیید مسئول ارشد و توضیح وضعیت پیگیری مجاز است. سکوت یا عدم پاسخ طرف مقابل به‌عنوان پذیرش اتهام معرفی نمی‌شود.

پاسخ مرتبطی که پس از انتشار دریافت می‌شود، بررسی و در جایگاهی متناسب به مطلب افزوده یا به آن پیوند داده می‌شود.

۱. تفاوت اصلاح و به‌روزرسانی

اصلاح برای رفع اطلاعات نادرست یا بازنمایی گمراه‌کننده انجام می‌شود.

به‌روزرسانی برای افزودن تحول یا اطلاعات تازه به خبری است که هنگام انتشار، بر اساس شواهد موجود درست گزارش شده است.

ویرایش نگارشی شامل تغییرهایی است که معنای خبر را عوض نمی‌کنند.

دریافت اطلاعات تازه بهانه‌ای برای نامیدن یک خطای قبلی به‌عنوان «به‌روزرسانی» نیست.

یادداشت اصلاح روشن می‌کند چه چیزی نادرست بوده و چه چیزی جایگزین آن شده است. عبارت کلی «خبر اصلاح شد» برای تغییرهای مهم کافی نیست.

اگر اصلاح به ادعای اصلی تیتر یا نتیجه خبر مربوط باشد، یادداشت در ابتدای مطلب یا نزدیک بخش اصلاح‌شده و در جایگاهی قابل مشاهده قرار می‌گیرد.

۳. اصلاح در همه بازنمایی‌های مرتبط

اصلاح یک خبر، متناسب با خطا، تیتر نمایشی، لید، متن، تصویر، عنوان صفحه، توضیحات متا است و عنوان‌های اشتراک‌گذاری را در بر می‌گیرد.

در حساب‌های رسمی شبکه‌های اجتماعی و دیگر کانال‌های تحت کنترل خبرآنلاین، خطای منتشرشده متناسب با امکانات هر کانال اصلاح یا با توضیح روشن پیگیری می‌شود. تغییر نمایش در سامانه‌های خارج از کنترل خبرآنلاین ممکن است فوری نباشد.

نسخه پیشین تغییرهای مهم، زمان تغییر و مسئول تأیید آن در سوابق داخلی تحریریه ثبت می‌شود. این سوابق موجب بازنشر عمومی اطلاعات خصوصی یا آسیب‌زا نیست.

۴. حذف و پس‌گرفتن خبر

حذف کامل مطلب، جایگزین معمول اصلاح نیست. اگر موضوع با اصلاح و توضیح قابل رفع باشد، اصل بر حفظ مطلب همراه با اصلاحیه است.

حذف یا پس‌گرفتن خبر در مواردی مانند بی‌اعتباری بنیادی محتوا، الزام قانونی، نقض جدی حریم خصوصی یا خطر آسیب قابل توجه، با تصمیم سردبیر انجام می‌شود.

تا حد ممکن، توضیحی روشن درباره حذف یا پس‌گرفتن مطلب در نشانی اصلی باقی می‌ماند؛ مگر آنکه خود این توضیح موجب تداوم آسیب یا نقض محدودیت قانونی شود.

۱. شفافیت هویت و منشأ محتوا

نام نویسنده یا واحد مسئول، نوع محتوا، تاریخ انتشار و زمان به‌روزرسانی مهم متناسب با قالب مطلب مشخص است.

محتوای بازنشرشده به منبع اصلی قابل شناسایی منتسب می‌شود و در صورت امکان به آن پیوند داده می‌شود. ترجمه، اقتباس، گزارش میدانی و تحلیل به‌گونه‌ای معرفی نمی‌شوند که منشأ یا روش تولید آن‌ها برای مخاطب گمراه‌کننده باشد.

در مطالب تخصصی، اطلاعات مرتبط درباره نویسنده یا کارشناس در دسترس مخاطب قرار می‌گیرد.

۲. تصاویر، ویدئو و محتوای کاربران

تصویر شاخص و محتوای چندرسانه‌ای رویداد، زمان، مکان یا اشخاص را نادرست بازنمایی نمی‌کند.

تصاویر آرشیوی، تزئینی، بازسازی‌شده یا ترکیبی به‌طور روشن معرفی می‌شوند.

برش و اصلاح فنی تصویر معنای خبری آن را تغییر نمی‌دهد.

افزودن یا حذف عناصر مؤثر تصویر خبری، به شکلی که واقعیت را تحریف کند، ممنوع است.

تصویر نامرتبط وقوع رویدادی را القا نمی‌کند که متن آن را تأیید نمی‌کند.

عنوان و زیرنویس ویدئو به محتوای اصلی وفادار است.

پیش از انتشار محتوای ارسالی کاربران، اصالت، زمان، مکان، زمینه، حقوق استفاده و احتمال آسیب بررسی می‌شود. فرستنده محتوا لزوماً مالک حقوق آن نیست و رضایت او به‌تنهایی همه ملاحظات مربوط به افراد حاضر در تصویر را برطرف نمی‌کند.

۳. علم، سلامت و داده

خبرهای علمی و سلامت به منابع تخصصی معتبر متکی هستند. نوع مطالعه، محدودیت‌های مهم و مقدماتی‌بودن یافته‌ها به زبان قابل فهم توضیح داده می‌شوند.

همبستگی رابطه علت‌ومعلولی معرفی نمی‌شود و نتیجه پژوهش آزمایشگاهی یا حیوانی بدون پشتوانه به درمان قطعی انسان تعمیم نمی‌یابد.

در گزارش داده‌ها، منبع، دوره زمانی، جامعه مورد بررسی، واحد اندازه‌گیری و مبنای مقایسه روشن است. نمودار و انتخاب اعداد نیز برداشت مخاطب را تحریف نمی‌کنند.

۴. حوادث، بحران‌ها و پرونده‌های قضایی

در پوشش بحران‌ها، کاهش آسیب و انتشار اطلاعات کاربردیِ بررسی‌شده اولویت دارد. آمار تلفات، هویت قربانیان و علت حادثه بر حدس یا رقابت رسانه‌ای استوار نیست.

انتشار تصاویر تکان‌دهنده نیازمند ضرورت خبری روشن، بررسی سردبیری و هشدار مناسب است.

در خبرهای قضایی، مرحله رسیدگی و منبع اطلاعات مشخص است. تیتر ادعا یا اتهام را به حکم قطعی تبدیل نمی‌کند.

۵. تبلیغات و محتوای دارای حمایت مالی

محتوای تجاری از محتوای مستقل تحریریه قابل تشخیص است. برچسب آن روشن و در ابتدای صفحه قرار می‌گیرد و در معرفی‌های تحت کنترل خبرآنلاین نیز ماهیت تجاری آن پنهان نمی‌شود.

«تبلیغ» یا «همکاری تجاری»

برچسب تبلیغاتی، مجوز انتشار ادعای فریبنده یا بی‌پشتوانه نیست. تبلیغ‌دهنده حق تعیین تکلیف درباره انتشار، حذف یا جهت‌گیری محتوای مستقل تحریریه را ندارد.

۶. هوش مصنوعی

استفاده از هوش مصنوعی برای کارهایی مانند پیاده‌سازی مصاحبه، کمک به ترجمه، خلاصه‌سازی اولیه و تحلیل داده‌ها، با نظارت انسانی مجاز است.

در این استفاده‌ها:

نقل‌قول با صوت، تصویر یا متن اصلی تطبیق داده می‌شود.

آمار، نام‌ها، پیوندها و منابع پیشنهادی مستقلاً بررسی می‌شوند.

ساخت نقل‌قول، سند، منبع یا آمار و عرضه آن به‌عنوان واقعیت ممنوع است.

اطلاعات محرمانه منابع و داده‌های شخصی بدون مجوز و ارزیابی حفاظت اطلاعات وارد ابزارهای بیرونی نمی‌شوند.

تصویر، صوت یا ویدئوی مصنوعی به‌عنوان ثبت واقعی رویداد عرضه نمی‌شود.

استفاده مؤثر از محتوای مصنوعی یا تولید ماشینی، هرگاه بر برداشت مخاطب از اصالت یا روش تولید اثر بگذارد، افشا می‌شود.

استفاده صرفاً ابزاری، مانند اصلاح املایی، نیازمند برچسب عمومی جداگانه نیست. مسئولیت نهایی محتوا در همه موارد با تحریریه است.

۷. شبکه‌های اجتماعی

حساب‌های رسمی خبرآنلاین مشمول همان اصول دقت، انتساب، انصاف و اصلاح خطا هستند. کوتاه‌بودن قالب، مجوز حذف شرط تعیین‌کننده یا گمراه‌کننده‌ترکردن تیتر نیست.

اعضای تحریریه در حساب‌های شخصی نیز محرمانگی منابع و اطلاعات داخلی را رعایت می‌کنند، از توهین و افترا می‌پرهیزند و میان دیدگاه شخصی و موضع رسمی رسانه تمایز می‌گذارند.

بخش هفتم: نمونه‌های آموزشی تیترنویسی

نمونه‌های زیر الگوی عملیاتی تحریریه خبرآنلاین در اصلاح تیترهای مبهم و جایگزینی آن‌ها را با تیترهای دقیق نشان می‌دهند. اعتبار هر تیتر پیشنهادی به وجود همان اطلاعات در منبع و متن خبر وابسته است.

بخش هشتم: نظارت، آموزش و پاسخ‌گویی

۱. برنامه نظارت

نظارت بر اجرای این سند در چهار سطح انجام می‌شود:

کنترل روزانه: بررسی پیش از انتشار و رسیدگی به گزارش‌های خطای فوری توسط دبیران و مسئول شیفت.

بازبینی هفتگی: بررسی نمونه‌ای از خروجی سرویس‌ها، موارد پرریسک و اصلاحات مهم در شورای سردبیری یا گروه تعیین‌شده از سوی آن.

ممیزی ماهانه آرشیو: بازبینی هدفمند مطالب قدیمی بر اساس گزارش مخاطبان، الگوهای خطا و موضوعات پرریسک.

بازنگری شش‌ماهه سیاست‌ها: ارزیابی کارآمدی دستورالعمل‌ها، نیازهای آموزشی و اصلاح سند.

دامنه نمونه‌گیری به‌گونه‌ای تعیین می‌شود که تمامی سرویس‌های خبری با اولویت سرویس‌های پرترافیک و پرریسک را پوشش دهد.

۲. آموزش و رسیدگی به تخلف

این سند در اختیار همه همکاران مرتبط قرار می‌گیرد و آموزش آن در آغاز همکاری و پس از تغییرهای مهم سیاست‌ها انجام می‌شود.

در صورت مشاهده تخلف، اولویت نخست اصلاح محتوا و کاهش آسیب است. سپس علت خطا، از جمله ضعف آموزش، نقص فرآیند، فشار کاری یا بی‌اعتنایی به دستورالعمل، بررسی می‌شود.

اقدام متناسب می‌تواند شامل آموزش مجدد، افزایش نظارت، محدودکردن اختیار انتشار یا رسیدگی اداری مطابق مقررات داخلی باشد.

۴. ارتباط با تحریریه

مخاطبان، کارشناسان و نهادها می‌توانند گزارش خطا، درخواست اصلاح یا جوابیه را از طریق بخش «ارتباط با ما» یا نشانی‌های زیر ارسال کنند:

corrections@khabaronline.ir برای گزارش خطا و درخواست اصلاح

info@khabaronline.ir برای سایر مکاتبات

گزارش‌ها برای بررسی اولیه به دبیر سرویس مربوط ارجاع می‌شوند. موارد حساس، اختلافی یا حل‌نشده در اختیار سردبیر قرار می‌گیرند.

گزارش‌های مربوط به خطر فوری، نقض حریم خصوصی یا خطای مهم در اولویت‌اند. هدف عملیاتی تحریریه، بررسی اولیه گزارش‌های خطا حداکثر ظرف ۲۴ ساعت از دریافت است. در موارد محرز، اصلاح بلافاصله پس از بررسی اولیه اعمال می‌شود و چنانچه بررسی نیازمند استعلام یا زمان بیشتری باشد، وضعیت به فرستنده اطلاع داده می‌شود.

تعهد نهایی خبرآنلاین

خبرآنلاین انتشار خبر را مسئولیتی عمومی می‌داند. دقت، استقلال حرفه‌ای، انصاف، شفافیت و پاسخ‌گویی بر هیجان، شتاب و منفعت کوتاه‌مدت مقدم هستند.

انتشار این سند به معنای ادعای بی‌خطابودن نیست؛ تعهد به پیشگیری از خطا، پذیرش مسئولیت، اصلاح آشکار و بهبود مستمر است. مخاطبان حق دارند عملکرد خبرآنلاین را بر اساس همین اصول ارزیابی کنند و موارد عدول از آن را با تحریریه در میان بگذارند.

تغییرهای مهم این سیاست‌ها باید همراه با تاریخ و خلاصه تغییر در همین صفحه اعلام شوند.
